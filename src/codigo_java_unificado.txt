package Meteorologia;
import java.util.ArrayList;
import java.util.List;

public class SistemaMeteorologico {
    private List<ZonaGeografica> zonas;
    private List<EstacionMeteorologica> estaciones;
    private List<ResponsableTecnico> responsables;
    private List<EquipoRespuesta> equipos;

    public SistemaMeteorologico() {
        this.zonas = new ArrayList<>();
        this.estaciones = new ArrayList<>();
        this.responsables = new ArrayList<>();
        this.equipos = new ArrayList<>();
    }


        public void mostrarRankingEstaciones() {
            System.out.println(" Ranking de estaciones por volumetria");

            estaciones.stream()
                    .sorted((e1, e2) -> {
                        double suma1 = e1.getSensores().stream().flatMap(s -> s.getMediciones().stream()).mapToDouble(Medicion::getMilimetros).sum();
                        double suma2 = e2.getSensores().stream().flatMap(s -> s.getMediciones().stream()).mapToDouble(Medicion::getMilimetros).sum();
                        return Double.compare(suma2, suma1); // Mayor a menor
                    })
                    .forEach(e -> {
                        double total = e.getSensores().stream().flatMap(s -> s.getMediciones().stream()).mapToDouble(Medicion::getMilimetros).sum();
                        System.out.println("Estación: " + e.getNombre() + " | Acumulado Total: " + total + " mm");
                    });
        }


        public void mostrarInformeFallasYCalibracion() {
            System.out.println("\n--- INFORME DE FALLAS Y CALIBRACIÓN ---");
            for (EstacionMeteorologica est : estaciones) {
                for (Sensor sen : est.getSensores()) {
                    if (sen.getEstado() != EstadoOperativo.OPERATIVO) {
                        System.out.println("[FUERA DE SERVICIO / MANTENIMIENTO] Estación: " + est.getNombre() + " - Sensor ID: " + sen.getIdSensor());
                    }
                    if (sen.requiereCalibracion()) {
                        System.out.println("[REQUIERE CALIBRACIÓN (>12 meses)] Estación: " + est.getNombre() + " - Sensor ID: " + sen.getIdSensor());
                    }
                }
            }
        }

        // Getters
        public void agregarZona(ZonaGeografica z) { zonas.add(z); }
        public void agregarEstacion(EstacionMeteorologica e) { estaciones.add(e); }
        public void agregarResponsable(ResponsableTecnico r) { responsables.add(r); }
        public void agregarEquipo(EquipoRespuesta eq) { equipos.add(eq); }

        public List<ZonaGeografica> getZonas() { return zonas; }
        public List<EstacionMeteorologica> getEstaciones() { return estaciones; }
    }

package Meteorologia;
import java.time.LocalDateTime;
import java.util.List;
import java.util.ArrayList;
import static java.util.Locale.filter;

public class ZonaGeografica {
    //atributos
    private int idZona;
    private String nombre;
    private NivelVulnerabilidad nivelVulnerabilidad;
    private NivelAlerta nivelAlertaActual ;
    private List<EstacionMeteorologica> estaciones;
    private List<ValorHistorico> valoresHistoricos;
    private List<Alerta> alertas;
    private double umbralAmarilla;
    private double umbralNaranja;
    private double umbralRoja;

    //constructor
    public ZonaGeografica(int idZona,String nombre,NivelVulnerabilidad nivelVunerabilidad,NivelAlerta nivelAlertaActual,List<EstacionMeteorologica> estaciones,List<Alerta> alertas,List<ValorHistorico> valoresHistoricos,double umbralAmarilla,double umbralNaranja,double umbralRoja){
        this.idZona=idZona;
        this.nombre=nombre;
        this.nivelVulnerabilidad=nivelVunerabilidad;
        this.nivelAlertaActual=nivelAlertaActual;
        this.estaciones = (estaciones != null) ? estaciones : new ArrayList<>();
        this.valoresHistoricos = (valoresHistoricos != null) ? valoresHistoricos : new ArrayList<>();
        this.alertas = (alertas != null) ? alertas : new ArrayList<>();
        this.umbralAmarilla=umbralAmarilla;
        this.umbralNaranja=umbralNaranja;
        this.umbralRoja=umbralRoja;
    }
    //getters y setters
    public int getIdZona() {
        return idZona;
    }

    public void setIdZona(int idZona) {
        this.idZona = idZona;
    }
    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public NivelVulnerabilidad getNivelVulnerabilidad() {
        return nivelVulnerabilidad;
    }

    public void setNivelVulnerabilidad(NivelVulnerabilidad nivelVulnerabilidad) {
        this.nivelVulnerabilidad = nivelVulnerabilidad;
    }

    public NivelAlerta getNivelAlerta() {
        return nivelAlertaActual;
    }

    public void setNivelAlertaActual(NivelAlerta nivelAlertaActual) {

        this.nivelAlertaActual = nivelAlertaActual;
    }

    public List<EstacionMeteorologica> getEstaciones() {
        return estaciones;
    }

    public void setEstaciones(List<EstacionMeteorologica> estaciones) {
        this.estaciones = estaciones;
    }

    public List<ValorHistorico> getValoresHistoricos() {
        return valoresHistoricos;
    }

    public void setValoresHistoricos(List<ValorHistorico> valoresHistoricos) {
     this.valoresHistoricos = valoresHistoricos;
    }

    public List<Alerta> getAlertas() {
        return alertas;
    }

    public void setAlertas(List<Alerta> alertas) {
        this.alertas = alertas;
    }

    public double getUmbralAmarilla() {return umbralAmarilla;}

    public void setUmbralAmarilla(double umbralAmarilla) {this.umbralAmarilla = umbralAmarilla;}

    public double getUmbralNaranja() {return umbralNaranja;}

    public void setUmbralNaranja(double umbralNaranja) {this.umbralNaranja = umbralNaranja;}

    public double getUmbralRoja() {return umbralRoja;}

    public void setUmbralRoja(double umbralRoja) {this.umbralRoja = umbralRoja;}

    //metodos
    public double lluviaAcumulada24h(){
        if(estaciones == null  || estaciones.isEmpty()){
            return 0.0;
        }
        LocalDateTime limite24h = LocalDateTime.now().minusHours(24);
        double totalAcumulado =0.0;

        for (EstacionMeteorologica estacion : estaciones){
            if(estacion.estaOperativa()){
                for(Sensor sensor : estacion.getSensores()){
                    for (Medicion medicion : sensor.getMediciones()){
                        if(medicion.getFechaHora().isAfter(limite24h)){
                            totalAcumulado += medicion.getMilimetros();
                        }
                    }
                }
            }
        }
        return totalAcumulado;
    }

    public NivelAlerta evaluarRiesgo(){
        double acumulada = this.lluviaAcumulada24h();
        if(acumulada >= this.umbralRoja){
            return NivelAlerta.ROJA;
        } else if ( acumulada >= this.umbralNaranja) {
            return NivelAlerta.NARANJA;
        } else if (acumulada >= this.umbralAmarilla) {
            return NivelAlerta.AMARILLA;
        } else {
            return NivelAlerta.VERDE;
        }
    }
    public Alerta emitirAlerta(TipoEventoClimatico ev, EquipoRespuesta equipo) {
        NivelAlerta nuevoNivel = evaluarRiesgo();
        this.nivelAlertaActual = nuevoNivel;

        if ((nuevoNivel == NivelAlerta.NARANJA || nuevoNivel == NivelAlerta.ROJA) && equipo == null) {
            throw new IllegalArgumentException("Error: Una alerta de nivel " + nuevoNivel + " requiere la asignación obligatoria de un equipo de respuesta.");
        }

        int nuevoId = this.alertas.size() + 1;
        String desc = "Alerta emitida automáticamente por acumulación de lluvia en 24h: " + lluviaAcumulada24h() + " mm.";
        
        boolean atendida = (equipo != null);
        Alerta nuevaAlerta = new Alerta(nuevoId, LocalDateTime.now(), nuevoNivel, ev, desc, atendida, equipo);
        
        this.alertas.add(nuevaAlerta);
        return nuevaAlerta;
    }

    public Alerta registrarAlertaOperador(NivelAlerta nivel, TipoEventoClimatico evento, String descripcion, EquipoRespuesta equipo) {
        // Regla: Alertas de nivel NARANJA o ROJA requieren asignación obligatoria de equipo
        if ((nivel == NivelAlerta.NARANJA || nivel == NivelAlerta.ROJA) && equipo == null) {
            throw new IllegalArgumentException("Error: Las alertas de nivel " + nivel + " registradas por operador requieren un equipo de respuesta obligatorio.");
        }

        int nuevoId = this.alertas.size() + 1;
        boolean atendida = (equipo != null);
        
        Alerta nuevaAlerta = new Alerta(nuevoId, LocalDateTime.now(), nivel, evento, descripcion, atendida, equipo);
        
        // Se actualiza el nivel de alerta vigente de la zona y se añade a la lista
        this.nivelAlertaActual = nivel;
        this.alertas.add(nuevaAlerta);
        
        return nuevaAlerta;
    }
}

package Meteorologia;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.ArrayList;

public class Sensor {
    private int idSensor;
    private String marca;
    private String modelo;
    private double precision;
    private EstadoOperativo estado;
    private LocalDate ultimaCalibracion;
    private List<Medicion> mediciones;

    //constructor
    public Sensor(int idSensor, String marca, String modelo, double precision, EstadoOperativo estado, LocalDate ultimaCalibracion,List<Medicion> mediciones) {
        this.idSensor = idSensor;
        this.marca = marca;
        this.modelo = modelo;
        this.precision = precision;
        this.estado = estado;
        this.ultimaCalibracion = ultimaCalibracion;
        this.mediciones = (mediciones != null) ? mediciones : new ArrayList<>();
    }

    //getters y setters
    public int getIdSensor() {
        return idSensor;
    }

    public void setIdSensor(int idSensor) {
        this.idSensor = idSensor;
    }
    
    public String getMarca() {
        return marca;
    }

    public void setMarca(String marca) {
        this.marca = marca;
    }
    
    public String getModelo() {
        return modelo;
    }
    
    public void setModelo(String modelo) {
        this.modelo = modelo;
    }

    public double getPrecision() {
        return precision;
    }
    
    public void setPrecision(double precision) {
        this.precision = precision;
    }

    public EstadoOperativo getEstado() {
        return estado;
    }
    
    public void setEstado(EstadoOperativo estado) {
        this.estado = estado;
    }

    public LocalDate getUltimaCalibracion() {
        return ultimaCalibracion;
    }

    public void setUltimaCalibracion(LocalDate ultimaCalibracion) {
        this.ultimaCalibracion = ultimaCalibracion;
    }
    public List<Medicion> getMediciones() {
        return mediciones;
    }

    public void setMediciones(List<Medicion> mediciones) {
        this.mediciones = mediciones;
    }



    //metoodos
    public void validar(double mm){
        if(mm < 0 || mm > 500 ){
            throw new LecturaInvalidaException("Error: la lectura de " + mm + "mm esta fuera de rango permitido (0 - 500 mm)");
        }
    }

    public boolean requiereCalibracion(){
        if(this.ultimaCalibracion == null){
            return true;
        }
        LocalDate fechaLimite = this.ultimaCalibracion.plusMonths(12);
        return LocalDate.now().isAfter(fechaLimite);
    }

    public void registrar(double mm,LocalDateTime fh){
        if (this.estado != EstadoOperativo.OPERATIVO) {
            throw new IllegalStateException("No se pueden registrar mediciones en un sensor que no está operativo.");
        }
        validar(mm);
        boolean anomala =false;
        if(this.mediciones != null && !this.mediciones.isEmpty()){
            Medicion ultimaMedicion=this.mediciones.get(this.mediciones.size()-1);
            double diferencia = Math.abs(mm - ultimaMedicion.getMilimetros());
            if (diferencia > 100.0){
                anomala= true;
            }
        }
        Medicion nuevaMedicion = new Medicion(mm, fh ,anomala);
        this.mediciones.add(nuevaMedicion);
    }
}
